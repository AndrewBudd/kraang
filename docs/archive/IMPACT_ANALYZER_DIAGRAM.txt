BFS Dependency Traversal:
─────────────────────────
```
function get_dependent_facts(fact_id, max_depth=10):
    visited = set()
    queue = [(fact_id, 0)]
    
    while queue:
        current, depth = queue.pop(0)
        if current in visited or depth > max_depth:
            continue
            
        visited.add(current)
        
        for dependent in dependency_graph[current]:
            if dependent not in visited:
                queue.append((dependent, depth + 1))
    
    return visited - {fact_id}
```

```
function estimate_loc(artifact_type, fact_count, locations):
    base = {
        'code': 10,
        'doc': 5,
        'config': 3
    }[artifact_type]
    
    complexity = calculate_complexity(fact_count, locations)
    multiplier = {
        'trivial': 1.0,
        'simple': 1.5,
        'moderate': 2.0,
        'complex': 3.0
    }[complexity]
    
    return base * fact_count * multiplier
```

```
function assess_risk(dependent_count, file_impacts, broken_count):
    if any(impact.risk == CRITICAL for impact in file_impacts):
        return CRITICAL
    if dependent_count > 10:
        return CRITICAL
    if broken_count > 3:
        return HIGH
    if dependent_count > 5:
        return HIGH
    if broken_count > 1:
        return MEDIUM
    return LOW
```
